// Semantic error: parameter name used twice in function signature
import "ifj25" for Ifj
class Program {
    static duplicate(x, x) {  // x defined twice
        return x + x
    }
    
    static main() {
        var result
        result = duplicate(5, 10)
        __d = Ifj.write(result)
        __d = Ifj.write("\n")
    }
}
